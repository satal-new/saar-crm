// Saar Advisory CRM — Google Apps Script Backend // Paste this entire file into your Apps Script editor const SHEET_NAME = 'CRM_Data'; function doGet(e) { try { const action = e.parameter.action; if (action === 'load') { return loadData(); } return respond({ status: 'error', message: 'Unknown GET action' }); } catch (err) { return respond({ status: 'error', message: err.toString() }); } } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); if (payload.action === 'save') { return saveData(payload.data); } return respond({ status: 'error', message: 'Unknown POST action' }); } catch (err) { return respond({ status: 'error', message: err.toString() }); } } function saveData(data) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); // Create sheet if it doesn't exist if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } // Save the entire DB as a single JSON blob in cell A1 // This is the simplest and most reliable approach sheet.clearContents(); sheet.getRange('A1').setValue(JSON.stringify(data)); // Also write human-readable tables for easy viewing in Sheets writeReadableTables(ss, data); return respond({ status: 'ok', savedAt: new Date().toISOString() }); } function loadData() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { return respond({ status: 'empty' }); } const raw = sheet.getRange('A1').getValue(); if (!raw) { return respond({ status: 'empty' }); } const data = JSON.parse(raw); return respond(data); } function writeReadableTables(ss, data) { // --- Companies sheet --- writeSheet(ss, 'Companies', ['Name','Status','Priority','City','State','Country','Industry', 'Lead Source','Referred By','First Contact','Next Meeting','Next Action', 'Website','Tags','Retainer','Equity %'], (data.companies || []).map(c => { const f = c.financial || {}; return [ c.name, c.status, c.priority, c.city, c.state, c.country, c.industry, c.leadSource, c.referredBy, c.firstContact, c.nextMeeting, c.nextAction, c.website, (c.tags || []).join(', '), f.retainer ? f.retainerCurrency + ' ' + f.retainer : '', f.equity ? f.equity + '%' : '' ]; }) ); // --- Contacts sheet --- writeSheet(ss, 'Contacts', ['Company','Name','Title','Email','Phone','LinkedIn'], (data.contacts || []).map(ct => { const co = (data.companies || []).find(c => c.id === ct.companyId); return [co ? co.name : '', ct.name, ct.title, ct.email, ct.phone, ct.linkedin]; }) ); // --- Interactions sheet --- writeSheet(ss, 'Interactions', ['Company','Date','Time','Type','Notes'], (data.interactions || []) .sort((a, b) => new Date(b.date) - new Date(a.date)) .map(i => { const co = (data.companies || []).find(c => c.id === i.companyId); return [co ? co.name : '', i.date, i.time, i.type, i.notes]; }) ); // --- Projects sheet --- writeSheet(ss, 'Projects', ['Company','Project / Issue','Status','Description'], (data.projects || []).map(p => { const co = (data.companies || []).find(c => c.id === p.companyId); return [co ? co.name : '', p.title, p.status, p.description]; }) ); } function writeSheet(ss, name, headers, rows) { let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); } else { sheet.clearContents(); } // Write header row with styling const headerRange = sheet.getRange(1, 1, 1, headers.length); headerRange.setValues([headers]); headerRange.setFontWeight('bold'); headerRange.setBackground('#1a2e44'); headerRange.setFontColor('#ffffff'); // Write data rows if (rows.length > 0) { sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); } // Auto-resize columns for readability for (let i = 1; i <= headers.length; i++) { sheet.autoResizeColumn(i); } } function respond(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
